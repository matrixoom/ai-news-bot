
# 分库分表测试
测试下每个组大概有多少只股票数据，目前看最多是40左右
```sql
WITH base AS (
    SELECT
        symbol,
        instrument_type,
        substr(symbol, 1, instr(symbol, '.') - 1) AS symbol_code,
        substr(symbol, instr(symbol, '.') + 1) AS symbol_exchange
    FROM market_stock_instrument
    WHERE symbol IS NOT NULL
      AND instr(symbol, '.') > 0
),
shard_stat AS (
    SELECT
        CASE
            WHEN symbol_exchange = 'BJ'
                THEN symbol_exchange || '_' || substr(symbol_code, -1)
            ELSE
                symbol_exchange || '_' || substr(symbol_code, 4, 2)
        END AS shard_key,
        COUNT(*) AS instrument_count
    FROM base
    GROUP BY
        CASE
            WHEN symbol_exchange = 'BJ'
                THEN symbol_exchange || '_' || substr(symbol_code, -1)
            ELSE
                symbol_exchange || '_' || substr(symbol_code, 4, 2)
        END
)
SELECT
    shard_key,
    instrument_count
FROM shard_stat
ORDER BY shard_key;
```
